-- =========================================================
-- OBSERVATÓRIO ELEITORAL
-- Estrutura inicial do banco
-- =========================================================

-- ---------------------------------------------------------
-- 1. TABELA DE ELEIÇÕES
-- ---------------------------------------------------------

create table if not exists public.eleicoes (
    id bigint generated by default as identity primary key,
    ano integer not null,
    turno integer not null default 1,
    descricao text not null,
    status text not null default 'demonstracao'
        check (status in ('demonstracao', 'aguardando', 'ao_vivo', 'encerrada')),
    created_at timestamptz not null default now(),
    updated_at timestamptz not null default now(),

    constraint eleicoes_ano_check check (ano >= 1900 and ano <= 2200),
    constraint eleicoes_turno_check check (turno in (1, 2)),
    constraint eleicoes_unique unique (ano, turno)
);

-- ---------------------------------------------------------
-- 2. CARGOS
-- ---------------------------------------------------------

create table if not exists public.cargos (
    id bigint generated by default as identity primary key,
    nome text not null unique,
    created_at timestamptz not null default now()
);

-- ---------------------------------------------------------
-- 3. CANDIDATOS
-- ---------------------------------------------------------

create table if not exists public.candidatos (
    id bigint generated by default as identity primary key,
    cargo_id bigint not null references public.cargos(id) on delete restrict,
    numero integer not null,
    nome text not null,
    nome_urna text,
    partido text,
    created_at timestamptz not null default now(),

    constraint candidatos_unique unique (cargo_id, numero)
);

-- ---------------------------------------------------------
-- 4. LOCALIDADES / SEÇÕES
-- ---------------------------------------------------------

create table if not exists public.localidades (
    id bigint generated by default as identity primary key,
    uf char(2) not null,
    municipio text not null,
    zona integer,
    secao integer,
    created_at timestamptz not null default now(),

    constraint localidades_uf_check check (char_length(uf) = 2),
    constraint localidades_zona_check check (zona is null or zona > 0),
    constraint localidades_secao_check check (secao is null or secao > 0),

    constraint localidades_unique
        unique (uf, municipio, zona, secao)
);

-- ---------------------------------------------------------
-- 5. RESULTADOS
-- Um registro representa o total de votos de um candidato
-- naquela seção/localidade.
-- ---------------------------------------------------------

create table if not exists public.resultados (
    id bigint generated by default as identity primary key,

    eleicao_id bigint not null
        references public.eleicoes(id) on delete cascade,

    candidato_id bigint not null
        references public.candidatos(id) on delete cascade,

    localidade_id bigint not null
        references public.localidades(id) on delete cascade,

    votos bigint not null default 0,

    atualizado_em timestamptz not null default now(),

    constraint resultados_votos_check check (votos >= 0),

    constraint resultados_unique
        unique (eleicao_id, candidato_id, localidade_id)
);

-- ---------------------------------------------------------
-- 6. BOLETINS DE URNA
-- ---------------------------------------------------------

create table if not exists public.boletins_urna (
    id bigint generated by default as identity primary key,

    eleicao_id bigint not null
        references public.eleicoes(id) on delete cascade,

    localidade_id bigint not null
        references public.localidades(id) on delete cascade,

    arquivo text,
    sha256 text,

    status text not null default 'recebido'
        check (status in ('recebido', 'validado', 'divergente', 'erro')),

    recebido_em timestamptz not null default now(),

    dados_json jsonb,

    created_at timestamptz not null default now()
);

-- ---------------------------------------------------------
-- 7. REGISTRO DAS COLETAS
-- ---------------------------------------------------------

create table if not exists public.coletas (
    id bigint generated by default as identity primary key,

    origem text not null,
    arquivo text,

    sha256 text,

    status text not null default 'recebido'
        check (status in ('recebido', 'processado', 'erro', 'ignorado')),

    mensagem text,

    recebido_em timestamptz not null default now(),

    created_at timestamptz not null default now()
);

-- =========================================================
-- ÍNDICES
-- =========================================================

create index if not exists idx_resultados_eleicao
    on public.resultados(eleicao_id);

create index if not exists idx_resultados_candidato
    on public.resultados(candidato_id);

create index if not exists idx_resultados_localidade
    on public.resultados(localidade_id);

create index if not exists idx_resultados_atualizado
    on public.resultados(atualizado_em);

create index if not exists idx_localidades_uf
    on public.localidades(uf);

create index if not exists idx_localidades_municipio
    on public.localidades(municipio);

create index if not exists idx_boletins_eleicao
    on public.boletins_urna(eleicao_id);

create index if not exists idx_boletins_status
    on public.boletins_urna(status);

create index if not exists idx_coletas_status
    on public.coletas(status);

-- =========================================================
-- FUNÇÃO PARA UPDATED_AT
-- =========================================================

create or replace function public.atualizar_updated_at()
returns trigger
language plpgsql
as $$
begin
    new.updated_at = now();
    return new;
end;
$$;

drop trigger if exists trg_eleicoes_updated_at
on public.eleicoes;

create trigger trg_eleicoes_updated_at
before update on public.eleicoes
for each row
execute function public.atualizar_updated_at();

-- ---------------------------------------------------------
-- CANDIDATOS DEMO
-- ---------------------------------------------------------

insert into public.candidatos
    (cargo_id, numero, nome, nome_urna, partido)
select
    c.id,
    x.numero,
    x.nome,
    x.nome_urna,
    x.partido
from public.cargos c
cross join (
    values
        (10, 'Candidato A', 'CANDIDATO A', 'PARTIDO A'),
        (20, 'Candidato B', 'CANDIDATO B', 'PARTIDO B'),
        (30, 'Candidato C', 'CANDIDATO C', 'PARTIDO C'),
        (40, 'Candidato D', 'CANDIDATO D', 'PARTIDO D')
) as x(numero, nome, nome_urna, partido)
where c.nome = 'Presidente'
on conflict (cargo_id, numero) do nothing;

-- ---------------------------------------------------------
-- RESULTADOS DEMO
-- ---------------------------------------------------------

insert into public.resultados
    (eleicao_id, candidato_id, localidade_id, votos)

select
    e.id,
    c.id,
    l.id,

    case
        when c.numero = 10 then 132
        when c.numero = 20 then 98
        when c.numero = 30 then 41
        when c.numero = 40 then 12
        else 0
    end

from public.eleicoes e

join public.candidatos c
    on c.cargo_id = (
        select id
        from public.cargos
        where nome = 'Presidente'
        limit 1
    )

cross join public.localidades l

where e.ano = 2026
  and e.turno = 1

on conflict (eleicao_id, candidato_id, localidade_id)
do update set
    votos = excluded.votos,
    atualizado_em = now();

do $$
begin

    if not exists (
        select 1
        from pg_publication_tables
        where pubname = 'supabase_realtime'
          and schemaname = 'public'
          and tablename = 'eleicoes'
    ) then
        alter publication supabase_realtime
        add table public.eleicoes;
    end if;

    if not exists (
        select 1
        from pg_publication_tables
        where pubname = 'supabase_realtime'
          and schemaname = 'public'
          and tablename = 'resultados'
    ) then
        alter publication supabase_realtime
        add table public.resultados;
    end if;

    if not exists (
        select 1
        from pg_publication_tables
        where pubname = 'supabase_realtime'
          and schemaname = 'public'
          and tablename = 'boletins_urna'
    ) then
        alter publication supabase_realtime
        add table public.boletins_urna;
    end if;

end $$;

-- =========================================================
-- TESTE FINAL
-- =========================================================

select
    e.ano,
    e.turno,
    e.status,
    c.numero,
    c.nome,
    sum(r.votos) as votos
from public.resultados r

join public.eleicoes e
    on e.id = r.eleicao_id

join public.candidatos c
    on c.id = r.candidato_id

group by
    e.ano,
    e.turno,
    e.status,
    c.numero,
    c.nome

order by votos desc;
